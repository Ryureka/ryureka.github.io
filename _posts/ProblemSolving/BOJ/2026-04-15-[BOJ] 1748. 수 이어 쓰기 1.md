### 풀이
~~~ java
import java.util.Scanner;

public class Main {
	public static void main(String[] args) {
		Scanner sc=new Scanner(System.in);
		
		int x=sc.nextInt();
		int xCopy=x;
		
		int cnt=1;
		while(xCopy/10!=0) {
			cnt++;
			xCopy/=10;
		}
		
		int num=1; //
		int num2=9;
		int sum=0;
		for (int i = 1; i < cnt; i++) {
			num*=10;
			sum+=num2*i;
			num2*=10;
		}
		System.out.println(cnt*(x-num+1)+sum);				
	}
}
~~~
