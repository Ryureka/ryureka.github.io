### 풀이
~~~ java
import java.util.Scanner;

public class Main {
	public static void main(String[] args) {
		Scanner sc=new Scanner(System.in);
		int N=sc.nextInt();
		int i,k,t,cnt=1;
		int a[]=new int [3];
		for(i=1;i<=N;i++) {
			if(0<i && i<100) cnt=i;
			else if(i==1000) break;
			else{
				k=0; t=i;
				while(t>0) {
					a[k]=t%10;
					t/=10; k++;
				}
				if(a[0]-a[1]==a[1]-a[2]) cnt++;
			}
		}
		System.out.println(cnt);
	}
}
~~~
