### 풀이
~~~ java
import java.util.Scanner;

public class Main{
	public static void main(String [] args) {
		Scanner sc=new Scanner(System.in);
		int input = sc.nextInt();
		
		for(int i=0;i<2*input-1;i++) 
		{
			if(i<input) 
			{
				for(int nblk=0;nblk<i;nblk++)
				{
					System.out.print(" ");
				}
				for(int j=0;j<2*input-1-2*i;j++)
				{
					System.out.print("*");
				}
				System.out.print("\n");
			}
			else
			{
				for(int nblk=0;nblk<2*input-1-(i+1);nblk++)
				{
					System.out.print(" ");
				}
				for(int k=0;k<2*(i+1)-(2*input-1);k++)
				{
					System.out.print("*");
				}
				System.out.print("\n");
			}
		}
	}
}
~~~
