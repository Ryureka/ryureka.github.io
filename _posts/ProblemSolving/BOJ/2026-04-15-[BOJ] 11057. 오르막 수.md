### 풀이
~~~ java
import java.util.Scanner;

public class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		
		int N = sc.nextInt();		
		int[][] dp = new int[N+1][10];
		int mod = 10007;
		
		for(int i = 0; i < 10; i++) {
			dp[1][i] = 1;
		}
		
		
		for(int n = 2; n <= N; n++) {
			for(int i = 0; i < 10; i++) {
				for(int j = i; j < 10; j++) {
					dp[n][i] += dp[n - 1][j];					
					dp[n][i] %= mod;
				}
			}
		}
		
		int sum = 0;
		for(int i = 0; i < 10; i++) {
			sum += dp[N][i];
			sum %= mod;
		}
		
		System.out.println(sum);		
	}
}
~~~
