### 풀이
~~~ java
import java.util.Scanner;

public class Main {
	static Scanner sc=new Scanner(System.in);
	
	public static void main(String[] args) {
		int N=sc.nextInt();
		int cnt=0;
		for (int i = 0; i < N; i++) {
			String s=sc.next();
			if(check(s)) cnt++;
		}
		System.out.println(cnt);
	}
	
	public static boolean check(String s) {
		int size=s.length();
		int used[]=new int [27];
		for (int i = 0; i < size; i++) {
			int alphaNum=s.charAt(i)-'a';
			if(used[alphaNum]==0 || used[alphaNum]==i) {
				used[alphaNum]=i+1;
			}else {
				return false;
			}
		}
		return true;
	}
}
~~~
