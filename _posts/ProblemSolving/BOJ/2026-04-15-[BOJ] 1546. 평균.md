### 풀이
~~~ java
import java.util.Scanner;

public class Main {
	static Scanner sc=new Scanner(System.in);
	public static void main(String[] args) {
		int N=sc.nextInt();
		int arr[]=new int [N];
		int max,idx=0;
		for (int i = 0; i < arr.length; i++) {
			arr[i]=sc.nextInt();
		}
		 
		for (int i = 1; i < arr.length; i++) {
			if(arr[idx]<arr[i]) {
				idx=i;
			}
		}
		max=arr[idx];
		
		double sum=0;
		for (int i = 0; i < arr.length; i++) {
			sum+=(double)arr[i]/max*100;
		}
		
		System.out.println(sum/N);
	}
}
~~~
