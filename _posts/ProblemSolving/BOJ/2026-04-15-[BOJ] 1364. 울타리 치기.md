### 풀이
~~~ java
import java.util.Scanner;

public class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		long answer = -1;
		long q = n/6;
		long r = n%6;		
		
		if(r == 0) answer = n + 6*((q-1)*q/2) + r + 1;
		else answer = n + 6*((q-1)*q/2) + q*r;
		System.out.println(answer);
	}
}
~~~
