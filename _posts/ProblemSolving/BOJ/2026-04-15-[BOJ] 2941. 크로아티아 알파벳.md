### 풀이
~~~ java
import java.util.Scanner;

public class Main {
	static Scanner sc=new Scanner(System.in);
	public static void main(String[] args) {
		String s=sc.next();
		s=s.replace("c=", "k").replace("c-","k").replace("dz=","k")
		.replace("d-", "k").replace("lj", "k").replace("nj", "k")
		.replace("s=", "k").replace("z=", "k");
		System.out.println(s.length());
	}
}
~~~
