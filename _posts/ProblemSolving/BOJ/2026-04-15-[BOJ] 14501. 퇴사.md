### 풀이
~~~ java
import java.util.Scanner;

public class Main {
	static int MAX=Integer.MIN_VALUE;
	public static void main(String[] args) {
		Scanner sc=new Scanner(System.in);
		int N=sc.nextInt();
		
		int T[]=new int[N];
		int P[]=new int[N];
		
		for (int i = 0; i < N; i++) {
			T[i]=sc.nextInt();
			P[i]=sc.nextInt();
		}
		
		go(0,T,P,N,0);
		
		System.out.println(MAX);
	}
	
	public static void go(int day,int T[],int P[],int N,int sum) {
		if(day>N) return;
		if(day==N) {
			if(sum>MAX) MAX=sum;
			return;
		}
		go(day+T[day],T,P,N,sum+P[day]);
		go(day+1,T,P,N,sum);
	}
}
~~~
