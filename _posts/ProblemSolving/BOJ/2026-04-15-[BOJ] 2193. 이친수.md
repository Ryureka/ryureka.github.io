### 풀이
~~~ java
import java.util.Scanner;

public class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int N = sc.nextInt();
		long[][] dp = new long[N+1][2];
		
		dp[1][0] = 0l;
		dp[1][1] = 1l;
		
		for(int i = 2; i <= N; i++) {
			dp[i][1] = dp[i-1][0];
			dp[i][0] = dp[i-1][0]+ dp[i-1][1];
		}
		
		System.out.println(dp[N][0] + dp[N][1]);
	}
}
~~~
