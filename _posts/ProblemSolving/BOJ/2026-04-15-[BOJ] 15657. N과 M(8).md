### 풀이
~~~ java
import java.util.Arrays;
import java.util.Scanner;

public class Main {
	public static void main(String[] args) {
		Scanner sc=new Scanner(System.in);
		int N=sc.nextInt();
		int M=sc.nextInt();
		int arr[]=new int[N];
		int arr2[]=new int[N];
		for(int i=0;i<N;i++) {
			arr[i]=sc.nextInt();
		}
		Arrays.sort(arr);
		go(arr,arr2,0,N,M,0);
	}
	
	public static void go(int arr[],int arr2[],int index,int N,int M,int start) {
		if(index==M) {
			StringBuilder sb=new StringBuilder();
			for(int i=0;i<M;i++) {
				sb.append(arr2[i]);
                sb.append(" ");
			}
			System.out.print(sb+"\n");
			return;
		}
		for(int i=start;i<N;i++) {
			arr2[index]=arr[i];
			go(arr,arr2,index+1,N,M,i);
		}
	}
}
~~~
