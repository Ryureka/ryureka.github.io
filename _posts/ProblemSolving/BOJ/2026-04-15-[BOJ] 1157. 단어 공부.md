### 풀이
~~~ java
import java.util.Scanner;

public class Main {
	static Scanner sc=new Scanner(System.in);
	
	public static void main(String[] args) {
		int used[]=new int [26]; 
		String s=sc.next();
		s=s.toLowerCase();
		int size=s.length();
		for (int i = 0; i < size; i++) {
			int alphaNum=s.charAt(i)-'a';
			used[alphaNum]++;
		}
		
		int idx=0;
		for (int i = 1; i < used.length; i++) {
			if(used[idx]<used[i]) {
				idx=i;
			}
		}
		
		int max=used[idx];
		boolean multi=false;
		int cnt=0;
		for (int i = 0; i < used.length; i++) {
			if(max==used[i]) {
				cnt++;
			}
			if(cnt==2) {
				multi=true;
				break;
			}
		}
		if(multi) {
			System.out.println("?");
		}else {
			char c=(char) (idx+'A');
			System.out.println(c);
		}
	}
}
~~~
