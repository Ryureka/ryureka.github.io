### 풀이
~~~ java
import java.util.Scanner;

public class Main {
	static Scanner sc=new Scanner(System.in);
	static int N;
	public static void main(String[] args) {
		N=sc.nextInt();
		
		int num=666;
		int cnt=0;
		String s;
		while(true) {
			s=Integer.toString(num);			
			if(s.contains("666")) {	
				cnt++;
			}			
			if(cnt==N) {
				System.out.println(num);
				break;
			}
			num++;
		}
	}
}
~~~
