## 풀이 구현
~~~ java
import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String A = sc.next();
        int n = A.length();
        int cnt = 0;
        for(int i = 0; i < n-3; i++) {
            for(int j = i+2; j < n-1; j++){
                if(A.charAt(i)=='(' && A.charAt(i+1)=='('
                    && A.charAt(j)==')' && A.charAt(j+1)==')'
                )
                cnt++;
            }
        }

        System.out.println(cnt);
    }
}
~~~
