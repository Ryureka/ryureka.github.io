## 풀이 구현
~~~ java
import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String A = sc.next();
        // Please write your code here.
        int n = A.length();
        int cnt = 0;
        for(int i = 0; i < n; i++) {
            for(int j = i+1; j < n; j++) {
                if(A.charAt(i) == '(' && A.charAt(j) == ')') cnt++;
            }
        }
        System.out.println(cnt);
    }
}
~~~
